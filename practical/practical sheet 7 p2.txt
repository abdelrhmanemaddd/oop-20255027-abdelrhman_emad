
import java.util.Scanner;


public class Student {
    private String studentname;
    private int studentID;
    private double grade;
    
    public Student(String studentname,int studentID,double grade) {
        this.studentname=studentname;
        this.studentID=studentID;
        this.grade=grade;     
    } 
    public Student() {
        String studentname="Unkown";
        int studentID=0;
        double grade=0.0;
    }
    public void ubdategrade(double newgrade) {
        grade=newgrade;
    }
    public boolean ispassed() {
      return  grade>=50;
    }
    public String Getstudentname() {
        return studentname;
    }
    public int GetstudentID() {
        return studentID;
    }
    public double Getgrade() {
        return grade;
    }  
}

public class Main {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        System.out.print("Enter the number of students: ");
        int n = input.nextInt();
        Student[] students = new Student[n];

        for (int i = 0; i < n; i++) {

            System.out.println("\nEnter information for student " + (i + 1));

            System.out.print("Enter student name: ");
            input.nextLine();
            String name = input.nextLine();

            System.out.print("Enter student ID: ");
            int id = input.nextInt();

            System.out.print("Enter student grade: ");
            double Grade= input.nextDouble();

            students[i] = new Student(name,id,Grade);
        }

        System.out.println("Student Information");

        for (int i = 0; i < n; i++) {

            System.out.println("Student Name: " +
                    students[i].Getstudentname());

            System.out.println("Student ID: " +
                    students[i].GetstudentID());

            System.out.println("Grade: " +
                    students[i].Getgrade());

            if (students[i].ispassed()) {
                System.out.println("Status: Passed");
            } else {
                System.out.println("Status: Failed");
            }
        }

    }
}
